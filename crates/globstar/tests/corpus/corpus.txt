x/{**,a}/**	x	no-match
x/{**,a}/**	x/	match
x/{**,a}/**	x/y	match
x/{**,a}/**	x/y/z	match
x/{**,a}/**	x/a	match
x/{**,a}/**	x/a/z	match

x/{**,a}/**	xy	no-match
# {a/**,b}/** is a/** union b/** — cover BOTH branches and the impossible root.
{a/**,b}/**	a	no-match
{a/**,b}/**	a/x	match
{a/**,b}/**	a/x/y	match
{a/**,b}/**	b	no-match
{a/**,b}/**	b/x	match
{a/**,b}/**	c	no-match
{a/**,b}/**	ab	no-match
# /{**,}/** is /** union //** — needs a leading separator.
/{**,}/**		no-match
/{**,}/**	/	match
/{**,}/**	/x/y	match
/{**,}/**	//x	match
/{**,}/**	x	no-match
# */{**,*}/** is */** union */*/** — needs a first segment then a separator.
*/{**,*}/**		no-match
*/{**,*}/**	a	no-match
*/{**,*}/**	a/b	match
*/{**,*}/**	a/b/c	match
# dot: the trailing `**` (and the leading `*` of */...) exposes segment
