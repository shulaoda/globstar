# ============================================================================
# corpus-err.txt — parse errors the other files do not already pin
# Format: PATTERN<TAB>-<TAB>err:Kind (the single-pattern row format)
# Kind is a GlobError variant name (see spec §6). Every pattern MUST
# raise that error at compile time.
# ============================================================================

## empty
	-	err:Empty
# §6.1: `!` and `!!` negate nothing.
!	-	err:Empty
!!	-	err:Empty
!!!	-	err:Empty

## trailing.backslash
foo\	-	err:TrailingBackslash
\	-	err:TrailingBackslash
a/b/c\	-	err:TrailingBackslash
# §5.1: the pattern ends inside a class's escape.
[a\	-	err:TrailingBackslash

## unterminated.class
[	-	err:UnterminatedClass
[abc	-	err:UnterminatedClass
foo[abc	-	err:UnterminatedClass
[a-	-	err:UnterminatedClass

## posix.first_bracket_literal_unterminated
# POSIX first-`]` rule (§8.5): a leading `]` is a member, so no closer
# is left (`[]` itself is in corpus-class.txt).
foo[]	-	err:UnterminatedClass
# fast-glob divergence: fast-glob tolerates `[]` as literal text (line 25
# upstream: the "webpack" pattern). We reject at parse via POSIX first-`]`.
\\\\/$^+?.()=!|{},[].*	-	err:UnterminatedClass

## unterminated.brace
{	-	err:UnterminatedBrace
{a	-	err:UnterminatedBrace
{a,b	-	err:UnterminatedBrace
{a,{b,c}	-	err:UnterminatedBrace
foo{a,b	-	err:UnterminatedBrace
# fast-glob divergence: fast-glob tolerates unterminated braces as literal
# text (lines 1253–1256 upstream). We reject at parse time. Also covers
# the negated `!{a,b` variant (same structure after outer `!` strip).
!{a,b	-	err:UnterminatedBrace

## invalid.range
# A first-position `]` starts a range too (§8.5), and `!` is below `]`;
# the plain reversed ranges are in corpus-class.txt and corpus-case.txt.
[]-!]	-	err:InvalidRange

## escaped.separator
# `\/` is rejected: a `/` can never appear inside a file name on any
# platform, so an escaped separator has no possible match (spec §4.2).
\/	-	err:EscapedSeparator
a\/b	-	err:EscapedSeparator
{a\/b,c}	-	err:EscapedSeparator
**/a\/b	-	err:EscapedSeparator
